class Solution:
    def countOfPairs(self, nums: List[int]) -> int:
        MOD = 10**9 + 7
        n = len(nums)
        if n == 0:
            return 0
        
        # Initialize previous dp array for i=0
        max_val = nums[0]
        prev = [0] * (max_val + 1)
        for v in range(max_val + 1):
            prev[v] = 1
        
        for i in range(1, n):
            current_max = nums[i]
            current = [0] * (current_max + 1)
            # Compute prefix sums of previous
            prefix_prev = [0] * (len(prev))
            prefix_prev[0] = prev[0]
            for j in range(1, len(prev)):
                prefix_prev[j] = (prefix_prev[j-1] + prev[j]) % MOD
            
            delta = nums[i] - nums[i-1]
            for v in range(current_max + 1):
                u_low = max(0, v - delta)
                u_high = min(v, nums[i-1])
                if u_low > u_high:
                    current[v] = 0
                else:
                    if u_low == 0:
                        sum_val = prefix_prev[u_high]
                    else:
                        sum_val = (prefix_prev[u_high] - prefix_prev[u_low - 1]) % MOD
                    current[v] = sum_val
            
            prev = current
        
        # Sum all possible values in the last step
        total = sum(prev) % MOD
        return total

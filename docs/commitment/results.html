from typing import List

class Solution:
    def maxEnergyBoost(self, energyDrinkA: List[int], energyDrinkB: List[int]) -> int:
        # Initialize the maximum energy for the first hour with either drink
        maxA = energyDrinkA[0]
        maxB = energyDrinkB[0]
        
        # Iterate through the rest of the hours
        for i in range(1, len(energyDrinkA)):
            # Compute the new maxA and maxB
            new_maxA = max(maxA + energyDrinkA[i], maxB + energyDrinkA[i])
            new_maxB = max(maxB + energyDrinkB[i], maxA + energyDrinkB[i])
            
            # Update the maxA and maxB for the next iteration
            maxA, maxB = new_maxA, new_maxB
        
        # The answer is the maximum of the two possible final states
        return max(maxA, maxB)

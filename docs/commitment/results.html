from typing import List
import heapq

class Solution:
    def findSafeWalk(self, grid: List[List[int]], health: int) -> bool:
        m, n = len(grid), len(grid[0])
        
        # Check if starting cell is unsafe with health 1
        if grid[0][0] == 1 and health == 1:
            return False
        
        # Initial health after starting cell
        initial_health = health - grid[0][0]
        if initial_health <= 0:
            return False
        
        # Initialize max_health matrix
        max_health = [[-1 for _ in range(n)] for _ in range(m)]
        max_health[0][0] = initial_health
        
        # Priority queue (max-heap simulated with negative values)
        heap = []
        heapq.heappush(heap, (-initial_health, 0, 0))
        
        # Directions: up, right, down, left
        directions = [(-1, 0), (0, 1), (1, 0), (0, -1)]
        
        while heap:
            current_health_neg, i, j = heapq.heappop(heap)
            current_health = -current_health_neg
            
            # Check if we reached the destination
            if i == m - 1 and j == n - 1:
                return True
            
            for di, dj in directions:
                ni, nj = i + di, j + dj
                if 0 <= ni < m and 0 <= nj < n:
                    new_health = current_health - grid[ni][nj]
                    if new_health >= 1:
                        if max_health[ni][nj] < new_health:
                            max_health[ni][nj] = new_health
                            heapq.heappush(heap, (-new_health, ni, nj))
        
        return False
